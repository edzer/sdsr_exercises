def nb_summary(g):
    print("Number of regions:", g.n)
    print("Number of nonzero links:", int(g.cardinalities.sum()))
    print("Average number of links:", round(g.cardinalities.mean(), 6))
    print("Disjoint connected subgraphs:", g.n_components)

# Moran I statistic       Expectation          Variance 
#        0.22384238       -0.16666667        0.04578844
def moran_test(x, g):
    """Moran's I test under randomisation, row-standardised weights, one-sided
    (alternative "greater"), like spdep::moran.test()"""
    m = esda.Moran(np.asarray(x), g.transform("r").to_W(), permutations=0)
    return pd.Series(dict(I=m.I, expectation=m.EI, variance=m.VI_rand,
                          z=m.z_rand, p_value=norm.sf(m.z_rand)))

Unlike the vignette, which uses distance neighbours up to 1300 m and creates a very dense representation, we choose k=4 k-nearest neighbours, then convert to symmetry (note that some point locations are duplicated, preventing the use of spatial indexing):

  • Python
(pr_nb_k4s <- spdep::knn2nb(spdep::knearneigh(properties, k=4), sym=TRUE, row.names=properties$id))
# Warning in spdep::knearneigh(properties, k = 4): knearneigh:
# identical points found
# Warning in spdep::knearneigh(properties, k = 4): knearneigh:
# kd_tree not available for identical points
# Warning in spdep::knn2nb(spdep::knearneigh(properties, k = 4), sym
# = TRUE, : neighbour object has 5 sub-graphs
# Neighbour list object:
# Number of regions: 1000 
# Number of nonzero links: 5874 
# Percentage nonzero weights: 0.5874 
# Average number of links: 5.874 
# 5 disjoint connected subgraphs
# k = 4 nearest neighbours, made symmetric. There are only 575 distinct locations
# for the 1000 properties, and libpysal's Graph.build_knn() refuses coincident points
# (as does the kd-tree in spdep, which falls back to brute force); here ties between
# identical locations are broken by row order, as the brute-force search in spdep does
xy = np.column_stack([properties.geometry.x, properties.geometry.y])
D = cdist(xy, xy)
np.fill_diagonal(D, np.inf)
nn = np.argsort(D, axis=1, kind="stable")[:, :4]
A = sparse.csr_matrix((np.ones(nn.size), (np.repeat(np.arange(len(xy)), 4), nn.ravel())),
                      shape=(len(xy), len(xy)))
pr_nb_k4s = graph.Graph.from_sparse(((A + A.T) > 0).astype(float), ids=properties["id"].to_numpy())
nb_summary(pr_nb_k4s)

def anova_chisq(small, big):
    """analysis of deviance table with a chi-square test, scale from the larger model,
    like anova(small, big, test = "Chisq") on gam objects"""
    ddf = small.df_resid - big.df_resid
    dev = small.ssr - big.ssr
    scale = big.ssr / big.df_resid
    return pd.DataFrame({"Resid. Df": [small.df_resid, big.df_resid],
                         "Resid. Dev": [small.ssr, big.ssr],
                         "Df": [np.nan, ddf], "Deviance": [np.nan, dev],
                         "Pr(>Chi)": [np.nan, chi2.sf(dev / scale, ddf)]})

Adding an upper level IID random effect to the base formula also improves the fit of the model substantially:

#   Resid. Df Resid. Dev      Df Deviance  Pr(>Chi)

# Signif. codes:  0 '***' 0.001 '**' 0.01 '*' 0.05 '.' 0.1 ' ' 1
# IID random effect for the municipality department: a random intercept, fitted with
# statsmodels' MixedLM (the lme4 analogue; REML for the estimates, ML for the
# likelihood ratio tests, as REML likelihoods are not comparable across fixed effects).
# Note that the R chunk uses the integer num_dep in s(num_dep, bs = "re"), which gives a
# random slope; the factor conversion only happens in exercise 16.4. With a factor, gam
# (method = "REML"), lme4 and MixedLM all give the same estimates.
def fit_iid(formula, reml):
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        return smf.mixedlm(formula, properties_in_dd,
                           groups=properties_in_dd["num_dep"]).fit(reml=reml)

def lrt_iid(fixed_only, formula):
    """likelihood ratio test of the random intercept (variance on the boundary of the
    parameter space, so the chi-square(1) p-value is halved)"""
    ols_ml, iid_ml = fixed_only, fit_iid(formula, reml=False)
    stat = 2 * (iid_ml.llf - ols_ml.llf)
    return pd.Series({"logLik (fixed only)": ols_ml.llf, "logLik (with IID)": iid_ml.llf,
                      "LR statistic": stat, "p-value": 0.5 * chi2.sf(stat, 1)})

pr_base_iid = fit_iid(f_pr, reml=True)
lrt_iid(pr_base, f_pr)
